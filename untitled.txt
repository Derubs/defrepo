
import tkinter as tk
from tkinter import ttk, messagebox
import json
import os

FILE_NAME = "polyclinic_data.json"

class PolyclinicApp:
    def __init__(self, root):
        self.root = root
        self.root.title("ИС 'Поликлиника' - Система управления приемом")
        self.root.geometry("600x700")

        # --- СТИЛИ ---
        style = ttk.Style()
        style.configure("TLabel", font=("Arial", 10))
        style.configure("Header.TLabel", font=("Arial", 12, "bold"))

        # --- ПЕРЕМЕННЫЕ (Соответствуют блокам A1, A2, A8) ---
        self.patient_name = tk.StringVar()
        self.passport = tk.StringVar()
        self.snils = tk.StringVar()
        self.policy = tk.StringVar()
        self.sick_leave_status = tk.StringVar(value="Закрыт")

        self.create_widgets()
        self.load_data()

    def create_widgets(self):
        # БЛОК A1, A2: Регистратура и Проверка документов
        reg_frame = ttk.LabelFrame(self.root, text=" Блоки A1, A2, A4: Регистратура ", padding=10)
        reg_frame.pack(fill="x", padx=10, pady=5)

        ttk.Label(reg_frame, text="ФИО Пациента:").grid(row=0, column=0, sticky="w")
        ttk.Entry(reg_frame, textvariable=self.patient_name, width=40).grid(row=0, column=1, pady=2)

        ttk.Label(reg_frame, text="Паспорт:").grid(row=1, column=0, sticky="w")
        ttk.Entry(reg_frame, textvariable=self.passport, width=40).grid(row=1, column=1, pady=2)

        ttk.Label(reg_frame, text="СНИЛС:").grid(row=2, column=0, sticky="w")
        ttk.Entry(reg_frame, textvariable=self.snils, width=40).grid(row=2, column=1, pady=2)

        ttk.Label(reg_frame, text="Полис ОМС:").grid(row=3, column=0, sticky="w")
        ttk.Entry(reg_frame, textvariable=self.policy, width=40).grid(row=3, column=1, pady=2)

        # БЛОК A3, A5, A6, A7: Прием врача и Анализы
        doctor_frame = ttk.LabelFrame(self.root, text=" Блоки A3, A5, A6, A7: Кабинет Врача ", padding=10)
        doctor_frame.pack(fill="both", expand=True, padx=10, pady=5)

        ttk.Label(doctor_frame, text="Жалобы и осмотр (Прием):").pack(anchor="w")
        self.treatment_text = tk.Text(doctor_frame, height=5, width=50)
        self.treatment_text.pack(pady=5)

        ttk.Label(doctor_frame, text="Назначенные анализы (Кровь/Моча/Кал):").pack(anchor="w")
        self.analysis_entry = tk.Entry(doctor_frame, width=50)
        self.analysis_entry.pack(pady=2)

        # БЛОК A8: Больничный лист (Продление/Закрытие)

        sick_frame = ttk.LabelFrame(self.root, text=" Блок A8: Выходные данные (Больничный лист) ", padding=10)
        sick_frame.pack(fill="x", padx=10, pady=5)

        ttk.Radiobutton(sick_frame, text="Открыть/Продлить", variable=self.sick_leave_status, value="Открыт/Продлен").pack(side="left", padx=20)
        ttk.Radiobutton(sick_frame, text="Закрыть", variable=self.sick_leave_status, value="Закрыт").pack(side="left", padx=20)

        # КНОПКИ УПРАВЛЕНИЯ
        btn_frame = ttk.Frame(self.root, padding=10)
        btn_frame.pack(fill="x")

        ttk.Button(btn_frame, text="Сохранить данные (Завершить прием)", command=self.save_data).pack(side="right", padx=5)
        ttk.Button(btn_frame, text="Очистить форму", command=self.clear_form).pack(side="right")

    def save_data(self):
        """Собирает все данные по схеме IDEF0 и сохраняет в JSON"""
        data = {
            "A1_A2_Registration": {
                "Name": self.patient_name.get(),
                "Passport": self.passport.get(),
                "SNILS": self.snils.get(),
                "Policy": self.policy.get()
            },
            "A3_A6_Treatment": self.treatment_text.get("1.0", tk.END).strip(),
            "A5_A7_Analysis": self.analysis_entry.get(),
            "A8_SickLeave": self.sick_leave_status.get()
        }

        with open(FILE_NAME, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=4)
        
        messagebox.showinfo("Успех", "Данные пациента успешно сохранены в систему.")

    def load_data(self):
        """Загружает данные последнего пациента"""
        if os.path.exists(FILE_NAME):
            try:
                with open(FILE_NAME, "r", encoding="utf-8") as f:
                    data = json.load(f)
                    
                    reg = data.get("A1_A2_Registration", {})
                    self.patient_name.set(reg.get("Name", ""))
                    self.passport.set(reg.get("Passport", ""))
                    self.snils.set(reg.get("SNILS", ""))
                    self.policy.set(reg.get("Policy", ""))
                    
                    self.treatment_text.delete("1.0", tk.END)
                    self.treatment_text.insert("1.0", data.get("A3_A6_Treatment", ""))
                    
                    self.analysis_entry.delete(0, tk.END)
                    self.analysis_entry.insert(0, data.get("A5_A7_Analysis", ""))
                    
                    self.sick_leave_status.set(data.get("A8_SickLeave", "Закрыт"))
            except Exception as e:
                print(f"Ошибка загрузки: {e}")

    def clear_form(self):
        self.patient_name.set("")
        self.passport.set("")
        self.snils.set("")
        self.policy.set("")
        self.treatment_text.delete("1.0", tk.END)
        self.analysis_entry.delete(0, tk.END)
        self.sick_leave_status.set("Закрыт")

if __name__ == "__main__":
    root = tk.Tk()
    app = PolyclinicApp(root)
    root.mainloop()
